구글 시트 실시간 연동 설정

학생들의 데이터가 실시간으로 수집되도록 웹 앱 URL을 등록하세요.

왜 데이터가 수집이 안 되었을까요? (필수 체크리스트)

  • 단순 구글 시트 주소(docs.google.com/...) 대신 스크립트 웹 앱 URL(script.google.com/.../exec)을 넣으셔야 합니다.
  • 앱스 스크립트 배포 시 액세스 권한을 반드시 "누구나 (Anyone)"로 선택하셔야 로그인 없이 전송됩니다.
구글 앱스 스크립트 전용 소스코드

구글 시트 메뉴 [확장 프로그램] -> [Apps Script]에 아래 코드를 붙여넣고 [새 배포]하세요.

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = data.slice(1);
  var result = rows.map(function(r) {
    return {
      className: String(r[0]),
      studentName: String(r[1]),
      total: Number(r[2]) || 0,
      YY: Number(r[3]) || 0,
      Yy: Number(r[4]) || 0,
      yy: Number(r[5]) || 0,
      yellow: Number(r[6]) || 0,
      green: Number(r[7]) || 0,
      timestamp: r[8]
    };
  });
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["학급", "이름", "총추출수", "YY", "Yy", "yy", "둥근완두(우성)", "주름진완두(열성)", "전송시간"]);
  }
  var values = sheet.getDataRange().getValues();
  var foundRow = -1;
  for (var i = 1; i < values.length; i++) {
    if (values[i][0] === data.className && values[i][1] === data.studentName) {
      foundRow = i + 1;
      break;
    }
  }
  var newRow = [
    data.className, data.studentName, data.total,
    data.YY, data.Yy, data.yy, data.yellow, data.green,
    new Date().toLocaleString("ko-KR")
  ];
  if (foundRow > 0) {
    sheet.getRange(foundRow, 1, 1, newRow.length).setValues([newRow]);
  } else {
    sheet.appendRow(newRow);
  }
  return ContentService.createTextOutput(JSON.stringify({result: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
